import glob
import os
import re
import pandas as pd


def find_global_max_across_csvs(folder_path, target_variable):
    max_val = float("-inf")
    max_file = None
    max_time = None

    # Retrieve all CSV files in the folder
    all_csvs = set(
        glob.glob(os.path.join(folder_path, "*.csv"))
        + glob.glob(os.path.join(folder_path, "*.CSV"))
    )

    # Filter specifically for segment files (e.g., 10_18_26_1.csv)
    csv_files = [
        f
        for f in all_csvs
        if len(os.path.basename(f).split("_")) >= 4
        or re.search(r"_\d+\.csv$", os.path.basename(f), re.IGNORECASE)
    ]

    if not csv_files:
        print(f"No matching segment CSV files found in {folder_path}")
        return None

    is_g_meter = any(g in target_variable.upper() for g in ["X AXIS", "Y AXIS"])

    for file_path in sorted(csv_files):
        filename = os.path.basename(file_path)
        try:
            # skiprows=[1] skips the Holley units row (e.g., [psi], [sec], [mph])
            df = pd.read_csv(file_path, encoding="latin-1", skiprows=[1])

            # Clean column names (strip leading/trailing whitespace)
            df.columns = df.columns.str.strip()

            # Dynamically look for the target channel
            match_col = next(
                (
                    col
                    for col in df.columns
                    if target_variable.upper() in col.upper()
                    and ("- F" in col or "FILTERED" in col.upper())
                ),
                None,
            )
            if not match_col:
                match_col = next(
                    (
                        col
                        for col in df.columns
                        if target_variable.upper() in col.upper()
                    ),
                    None,
                )

            if not match_col:
                continue

            df[match_col] = pd.to_numeric(df[match_col], errors="coerce")

            # Apply a light rolling smoothing window (5-point) specifically for G-meter channels
            # to kill jagged mechanical/electrical spikes while preserving actual acceleration curves.
            if is_g_meter:
                df[match_col] = df[match_col].rolling(window=5, center=True).mean()

            # Find time column dynamically
            time_col = next(
                (
                    col
                    for col in df.columns
                    if "time" in col.lower()
                    or "sec" in col.lower()
                    or "rtc" in col.lower()
                ),
                None,
            )
            if time_col:
                df[time_col] = pd.to_numeric(df[time_col], errors="coerce")

                # Ignore startup ramp/noise in the first segment file (_1.csv)
                if filename.lower().endswith("_1.csv"):
                    df = df[df[time_col] >= 2.0]

            # Find TPS column to use as a validity gate for G-meter peaks
            tps_col = next(
                (col for col in df.columns if "tps" in col.lower()), None
            )
            if tps_col:
                df[tps_col] = pd.to_numeric(df[tps_col], errors="coerce")

            # Apply specific filtering gates for G-meter variables to ignore shift/lift transients
            if is_g_meter:
                # 1. Require basic forward acceleration threshold (> 0.05G)
                df = df[df[match_col] > 0.05]

                # 2. TPS Gate: Real hard acceleration requires throttle to be planted (> 50% TPS)
                # This explicitly blocks post-lift shift spikes where TPS drops off.
                if tps_col:
                    df = df[df[tps_col] > 50.0]

            valid_df = df.dropna(subset=[match_col])
            if valid_df.empty:
                continue

            local_max_idx = valid_df[match_col].idxmax()
            local_max_val = valid_df.loc[local_max_idx, match_col]
            local_time = (
                valid_df.loc[local_max_idx, time_col]
                if time_col
                else "Unknown"
            )

            if local_max_val > max_val:
                max_val = local_max_val
                max_file = filename
                max_time = local_time

        except Exception as e:
            print(f"Error reading {filename}: {e}")

    return {"max_value": max_val, "file": max_file, "time": max_time}


if __name__ == "__main__":

    variable = "X Axis"

    print(f"Scanning segment files for maximum '{variable}'...")
    result = find_global_max_across_csvs(log_folder, variable)
